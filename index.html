churn_analysis.sql
SQL ANALYSIS 
WITH ranked_subscriptions AS (
  SELECT
    customer_id,
    end_date,
    mrr,
    ROW_NUMBER() OVER (
      PARTITION BY customer_id
      ORDER BY end_date DESC
    ) AS rn
  FROM subscriptions
)

SELECT
  c.customer_id,
  c.plan_type,
  rs.end_date,
  rs.mrr
FROM customers c
JOIN ranked_subscriptions rs
  ON c.customer_id = rs.customer_id
WHERE rs.rn = 1
  AND rs.end_date < CURRENT_DATE
ORDER BY rs.mrr DESC;

Demonstrates a CTE, JOIN and ROW_NUMBER() window function to identify the latest subscription record for each customer.

data_cleaning.py
PYTHON 
import pandas as pd

df = pd.read_csv("product_events.csv")

# Standardize column names
df.columns = (
    df.columns
      .str.strip()
      .str.lower()
      .str.replace(" ", "_")
)

# Convert event dates
df["event_date"] = pd.to_datetime(
    df["event_date"],
    errors="coerce"
)

# Remove duplicate events
df = df.drop_duplicates(
    subset=["customer_id",
            "event_date",
            "event_name"]
)

# Handle missing session duration
df["session_minutes"] = (
    df["session_minutes"]
      .fillna(0)
)
